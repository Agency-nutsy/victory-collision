"use client";

import { useEffect, useRef } from "react";
import { gsap } from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";
import { ClipboardList, Droplets, Sparkles, Shield, CheckCircle, type LucideIcon } from "lucide-react";

gsap.registerPlugin(ScrollTrigger);

const iconMap: Record<string, LucideIcon> = {
  ClipboardList,
  Droplets,
  Sparkles,
  Shield,
  CheckCircle,
};

interface ProcessStep {
  step: string;
  title: string;
  description: string;
  icon: string;
  videoUrl?: string;
  posterUrl?: string;
}

export default function ProcessTimeline({ steps }: { steps: ProcessStep[] }) {
  const containerRef = useRef<HTMLDivElement>(null);
  const scrollWrapperRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const isDesktop = window.matchMedia("(min-width: 1024px)").matches;
    if (!isDesktop) return;
    if (!containerRef.current || !scrollWrapperRef.current) return;

    const container = containerRef.current;
    const scrollWrapper = scrollWrapperRef.current;

    // Horizontal translation distance = how far to slide the track
    const getScrollDist = () => {
      const vw = window.innerWidth;
      const cardW = vw * 0.35;
      const gap = 32; // gap-8 = 2rem = 32px
      const leftPad = vw * 0.10;
      const n = steps.length;
      const totalW = leftPad + n * cardW + (n - 1) * gap;
      return totalW - vw;
    };

    // Pin length = scroll runway. Give each card one full viewport height.
    // e.g. 4 cards â†’ 3 Ã— innerHeight of vertical scroll to move through all cards.
    const getPinLength = () => (steps.length - 1) * window.innerHeight;

    const tween = gsap.to(scrollWrapper, {
      x: () => -getScrollDist(),
      ease: "none",
      scrollTrigger: {
        trigger: container,
        pin: true,
        scrub: 0.3,
        end: () => `+=${getPinLength()}`,
        invalidateOnRefresh: true,
      },
    });

    return () => {
      tween.kill();
      ScrollTrigger.getAll().forEach((t) => t.kill());
    };
  }, [steps]);

  if (!steps || steps.length === 0) return null;

  return (
    <section
      ref={containerRef}
      className="h-screen flex flex-col justify-center bg-primary text-base overflow-hidden relative z-30"
    >
      <div className="absolute inset-0 bg-primary opacity-50 pointer-events-none" />

      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 mb-12 relative z-10 w-full">
        <h2 className="text-sm font-bold tracking-widest text-accent uppercase mb-2">How It Works</h2>
        <h3 className="text-4xl md:text-5xl font-heading font-bold text-base">Our Signature Process</h3>
      </div>

      <div className="relative z-10 w-full overflow-hidden">
        <div
          ref={scrollWrapperRef}
          className="flex flex-nowrap pl-[10vw] pb-8 gap-8 hide-scrollbar"
        >
          {steps.map((step, idx) => {
            const Icon = iconMap[step.icon] || Sparkles;
            return (
              <div
                key={idx}
                className="w-[35vw] flex-shrink-0 flex flex-col gap-6"
              >
                {step.videoUrl ? (
                  <div className="relative rounded-2xl h-[380px] flex-shrink-0 border border-secondary/50 overflow-hidden bg-primary/80 backdrop-blur-md shadow-2xl">
                    <video
                      src={step.videoUrl}
                      poster={step.posterUrl}
                      autoPlay
                      loop
                      muted
                      playsInline
                      preload="none"
                      className="absolute inset-0 w-full h-full object-cover"
                    />
                  </div>
                ) : (
                  <div className="bg-primary/80 backdrop-blur-md rounded-2xl p-8 md:p-12 border border-secondary/50 flex flex-col justify-between group hover:border-[var(--secondary-color)] transition-colors h-[380px]">
                    <div>
                      <div className="flex justify-between items-start mb-12">
                        <span className="text-6xl font-black text-[#F5F1EA]/10 group-hover:text-[#F5F1EA]/20 transition-colors font-heading tracking-tighter">
                          {step.step}
                        </span>
                        <div className="w-16 h-16 rounded-full bg-primary/50 flex items-center justify-center text-secondary">
                          <Icon className="w-8 h-8" />
                        </div>
                      </div>
                      <h4 className="text-2xl font-bold font-heading mb-4 text-base">{step.title}</h4>
                      <p className="text-secondary text-lg leading-relaxed">{step.description}</p>
                    </div>
                  </div>
                )}

                {step.videoUrl && (
                  <div className="px-2">
                    <h4 className="text-2xl font-bold font-heading mb-2 text-base">{step.title}</h4>
                    <p className="text-secondary text-lg leading-relaxed">{step.description}</p>
                  </div>
                )}
              </div>
            );
          })}
        </div>
      </div>
    </section>
  );
}
